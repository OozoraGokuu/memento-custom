cmake_minimum_required(VERSION 3.16)
project(MementoXPPrototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(xp_core_tests core_tests.cpp)
add_test(NAME xp-prototype-core COMMAND xp_core_tests)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The XP prototype requires a 32-bit toolchain")
  endif()
  set(VLC_SDK "" CACHE PATH "Path to the extracted VLC sdk directory")
  if(NOT EXISTS "${VLC_SDK}/include/vlc/vlc.h")
    message(FATAL_ERROR "Set VLC_SDK to the official VLC 3.0.21 win32 sdk directory")
  endif()
  add_executable(memento-xp-prototype WIN32 main.cpp)
  target_include_directories(memento-xp-prototype PRIVATE "${VLC_SDK}/include")
  target_compile_definitions(memento-xp-prototype PRIVATE UNICODE _UNICODE WINVER=0x0501 _WIN32_WINNT=0x0501)
  target_link_libraries(memento-xp-prototype PRIVATE "${VLC_SDK}/lib/libvlc.lib" comctl32 comdlg32 shell32)
  if(MINGW)
    foreach(target memento-xp-prototype xp_core_tests)
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
      target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--major-subsystem-version,5,--minor-subsystem-version,1)
    endforeach()
    target_link_options(memento-xp-prototype PRIVATE -municode)
  endif()
endif()
